        for(int j=0;j<num[i].length;j++){
            if(num[i][j]!=0)
                sum++;
        }
    }

    //创建稀疏数组
    int[][] num1=new int[sum+1][3];

    for(int i=0;i<num.length;i++){
        for(int j=0;j<num[i].length;j++){
            if(num[i][j]!=0){

                num1[count][0]=i;
                num1[count][1]=j;
                num1[count][2]=num[i][j];
            }  
        }
